        // Audio Synthesizer via Web Audio API
        const audioCtx = new (window.AudioContext || window.webkitAudioContext)();

        function playSound(type) {
            try {
                if (audioCtx.state === 'suspended') {
                    audioCtx.resume();
                }
                const osc = audioCtx.createOscillator();
                const gain = audioCtx.createGain();
                osc.connect(gain);
                gain.connect(audioCtx.destination);

                if (type === 'click') {
                    osc.type = 'sine';
                    osc.frequency.setValueAtTime(600, audioCtx.currentTime);
                    osc.frequency.exponentialRampToValueAtTime(800, audioCtx.currentTime + 0.05);
                    gain.gain.setValueAtTime(0.15, audioCtx.currentTime);
                    gain.gain.linearRampToValueAtTime(0, audioCtx.currentTime + 0.05);
                    osc.start();
                    osc.stop(audioCtx.currentTime + 0.05);
                } else if (type === 'correct') {
                    osc.type = 'triangle';
                    osc.frequency.setValueAtTime(523.25, audioCtx.currentTime);
                    osc.frequency.setValueAtTime(659.25, audioCtx.currentTime + 0.08);
                    osc.frequency.setValueAtTime(783.99, audioCtx.currentTime + 0.16);
                    gain.gain.setValueAtTime(0.2, audioCtx.currentTime);
                    gain.gain.linearRampToValueAtTime(0, audioCtx.currentTime + 0.3);
                    osc.start();
                    osc.stop(audioCtx.currentTime + 0.3);
                } else if (type === 'wrong') {
                    osc.type = 'sawtooth';
                    osc.frequency.setValueAtTime(280, audioCtx.currentTime);
                    osc.frequency.linearRampToValueAtTime(140, audioCtx.currentTime + 0.25);
                    gain.gain.setValueAtTime(0.2, audioCtx.currentTime);
                    gain.gain.linearRampToValueAtTime(0, audioCtx.currentTime + 0.25);
                    osc.start();
                    osc.stop(audioCtx.currentTime + 0.25);
                } else if (type === 'win') {
                    const notes = [523.25, 659.25, 783.99, 1046.50];
                    notes.forEach((freq, idx) => {
                        const o = audioCtx.createOscillator();
                        const g = audioCtx.createGain();
                        o.connect(g);
                        g.connect(audioCtx.destination);
                        o.type = 'triangle';
                        o.frequency.value = freq;
                        g.gain.setValueAtTime(0.2, audioCtx.currentTime + idx * 0.1);
                        g.gain.linearRampToValueAtTime(0, audioCtx.currentTime + idx * 0.1 + 0.25);
                        o.start(audioCtx.currentTime + idx * 0.1);
                        o.stop(audioCtx.currentTime + idx * 0.1 + 0.25);
                    });
                }
            } catch (e) {
                console.log("Audio play error:", e);
            }
        }

            currentLang = lang;

            if (lang === 'ar') {

            const langLabels = { tn: 'Tounsi 🇹🇳', ar: 'العربية 🇵🇸', en: 'English 🇬🇧', fr: 'Français 🇫🇷' };

            const t = i18n[lang];

                renderQuestion();
            }
        }

        // Validate Player Name and Start Game
        function validateAndStart() {
            const input = document.getElementById('playerNameInput');
            const nameVal = input.value.trim();

            if (!nameVal) {
                playSound('wrong');
                input.focus();
                input.classList.add('border-rose-500', 'ring-2', 'ring-rose-500/50');
                setTimeout(() => {
                    input.classList.remove('border-rose-500', 'ring-2', 'ring-rose-500/50');
                }, 1500);
                return;
            }

            playerName = nameVal;
            playSound('click');
            startQuiz();
        }

        // Start Quiz Game
        function startQuiz() {
            currentQuestion = 0;
            score = 0;
            userAnswers = [];

            document.getElementById('startScreen').classList.add('hidden');
            document.getElementById('resultsScreen').classList.add('hidden');
            document.getElementById('quizScreen').classList.remove('hidden');

            document.getElementById('playerTag').textContent = `Player: ${playerName}`;

            renderPaginationPills();
            renderQuestion();
        }

        // Render Pagination Pills (1 - 20)
        function renderPaginationPills() {
            const container = document.getElementById('paginationPills');
            container.innerHTML = '';

            for (let i = 0; i < questionsData.length; i++) {
                const pill = document.createElement('div');
                pill.id = `pill-${i}`;
                pill.className = `w-7 h-7 md:w-8 md:h-8 rounded-full flex items-center justify-center font-bold text-xs transition-all border ${
                    i === currentQuestion 
                        ? 'bg-sky-500 text-white border-sky-300 scale-110 shadow-lg shadow-sky-500/40' 
                        : 'bg-slate-800 text-slate-400 border-slate-700'
                }`;
                pill.innerText = i + 1;
                container.appendChild(pill);
            }
        }

        // Render Active Question
        function renderQuestion() {
            const qData = questionsData[currentQuestion];

            // Badge & Counter
            document.getElementById('qBadge').textContent = `Q ${currentQuestion + 1} / 20`;
            document.getElementById('questionText').textContent = qData.q[currentLang];

            // Options Text & Realistic Unsplash Images
            document.getElementById('optAText').textContent = qData.optA[currentLang];
            document.getElementById('optAImg').src = qData.optA.img;

            document.getElementById('optBText').textContent = qData.optB[currentLang];
            document.getElementById('optBImg').src = qData.optB.img;

            resetOptionStyles();

            // Update Pagination Status
            for (let i = 0; i < questionsData.length; i++) {
                const pill = document.getElementById(`pill-${i}`);
                if (!pill) continue;

                if (i < userAnswers.length) {
                    const ans = userAnswers[i];
                    if (ans.isCorrect) {
                        pill.className = "w-7 h-7 md:w-8 md:h-8 rounded-full flex items-center justify-center font-bold text-xs bg-emerald-500 text-white border-emerald-400 shadow-md shadow-emerald-500/20";
                    } else {
                        pill.className = "w-7 h-7 md:w-8 md:h-8 rounded-full flex items-center justify-center font-bold text-xs bg-rose-500 text-white border-rose-400 shadow-md shadow-rose-500/20";
                    }
                } else if (i === currentQuestion) {
                    pill.className = "w-7 h-7 md:w-8 md:h-8 rounded-full flex items-center justify-center font-bold text-xs bg-sky-500 text-white border-sky-300 scale-110 shadow-lg shadow-sky-500/40 ring-2 ring-sky-400/30";
                } else {
                    pill.className = "w-7 h-7 md:w-8 md:h-8 rounded-full flex items-center justify-center font-bold text-xs bg-slate-800 text-slate-400 border-slate-700";
                }
            }

            // Scroll pill active view
            const activePill = document.getElementById(`pill-${currentQuestion}`);
            if (activePill) {
                activePill.scrollIntoView({ behavior: 'smooth', inline: 'center', block: 'nearest' });
            }
        }

        function resetOptionStyles() {
            const btnA = document.getElementById('optA');
            const btnB = document.getElementById('optB');

            btnA.className = "image-card group p-3 bg-slate-900/60 rounded-2xl border border-slate-700/80 hover:border-sky-400 flex flex-col items-center text-center transition";
            btnB.className = "image-card group p-3 bg-slate-900/60 rounded-2xl border border-slate-700/80 hover:border-sky-400 flex flex-col items-center text-center transition";

            btnA.disabled = false;
            btnB.disabled = false;
        }

        // Handle Answer Click
        function handleAnswer(choice) {
            const qData = questionsData[currentQuestion];
            const isCorrect = (qData.correct === 'BOTH') || (choice === qData.correct);

            if (isCorrect) {
                playSound('correct');
                score++;
            } else {
                playSound('wrong');
            }

            userAnswers.push({ choice, isCorrect });

            const selectedBtn = choice === 'A' ? document.getElementById('optA') : document.getElementById('optB');
            const otherBtn = choice === 'A' ? document.getElementById('optB') : document.getElementById('optA');

            selectedBtn.disabled = true;
            otherBtn.disabled = true;

            if (isCorrect) {
                selectedBtn.className = "image-card group p-3 bg-emerald-950/80 rounded-2xl border-2 border-emerald-400 flex flex-col items-center text-center shadow-lg shadow-emerald-500/20";
            } else {
                selectedBtn.className = "image-card group p-3 bg-rose-950/80 rounded-2xl border-2 border-rose-500 flex flex-col items-center text-center shadow-lg shadow-rose-500/20";
                
                // Highlight correct option
                if (qData.correct === 'A') {
                    document.getElementById('optA').className = "image-card group p-3 bg-emerald-950/60 rounded-2xl border-2 border-emerald-400/80 flex flex-col items-center text-center";
                } else if (qData.correct === 'B') {
                    document.getElementById('optB').className = "image-card group p-3 bg-emerald-950/60 rounded-2xl border-2 border-emerald-400/80 flex flex-col items-center text-center";
                }
            }

            // Next question transition delay
            setTimeout(() => {
                currentQuestion++;
                if (currentQuestion < questionsData.length) {
                    renderQuestion();
                } else {
                    showFinalResults();
                }
            }, 850);
        }

        // Send completed quiz to the server so the owner can view it at /userlist.
        async function saveToUserList() {
            const submission = {
                name: playerName,
                score: score,
                totalQuestions: questionsData.length,
                answers: userAnswers.map((answer, index) => ({
                    questionNumber: index + 1,
                    choice: answer.choice,
                    correct: answer.isCorrect
                })),
                completedAt: new Date().toISOString()
            };
            try {
                const response = await fetch('/api/submit', {
                    method: 'POST',
                    headers: { 'Content-Type': 'application/json' },
                    body: JSON.stringify(submission)
                });
                if (!response.ok) throw new Error('Server could not save the result');
            } catch (error) {
                console.error('Could not save result on server:', error);
                alert('Ma najjemnech nsajjel el résultat. Thabet mel connexion w 3awed.');
            }
        }

        // Show Final Score Screen

                messageBox.className = "my-6 p-5 rounded-2xl border bg-gradient-to-r from-amber-500/20 to-yellow-500/20 border-amber-400/50 text-amber-200 text-lg md:text-xl font-bold font-adapt shadow-lg shadow-amber-500/10";

                triggerConfettiCelebration();
