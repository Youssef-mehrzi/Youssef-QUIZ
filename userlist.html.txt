  <script>
    async function loadList() {
      const status = document.getElementById('status');
      const results = document.getElementById('results');

      status.textContent = 'جاري تحميل النتائج...';
      results.innerHTML = '';

      try {
        const res = await fetch('/api/userlist', {
          cache: 'no-store'
        });

        if (res.status === 401) {
          status.textContent =
            'يلزمك تسجّل الدخول بحساب المدير.';
          return;
        }

        if (!res.ok) {
          throw new Error('HTTP ' + res.status);
        }

        const list = await res.json();

        status.textContent =
          'عدد المشاركات: ' + list.length;

        if (!list.length) {
          results.innerHTML =
            '<div class="card">ما فما حتى نتيجة توّة.</div>';
          return;
        }

        list.slice().reverse().forEach((u) => {
          const card = document.createElement('section');
          card.className = 'card';

          const title = document.createElement('h2');
          title.textContent = u.name || 'بدون اسم';

          card.appendChild(title);

          const summary = document.createElement('p');

          summary.textContent =
            `النتيجة: ${u.score} / ${u.totalQuestions}` +
            ` — التاريخ: ${u.completedAt || 'غير معروف'}`;

          card.appendChild(summary);

          const wrap = document.createElement('div');
          wrap.className = 'scroll';

          const table = document.createElement('table');

          const thead = document.createElement('thead');

          thead.innerHTML =
            '<tr>' +
            '<th>السؤال</th>' +
            '<th>الإجابة</th>' +
            '<th>صحيحة؟</th>' +
            '</tr>';

          table.appendChild(thead);

          const tbody = document.createElement('tbody');

          (u.answers || []).forEach(a => {
            const tr = document.createElement('tr');

            [
              a.questionNumber,
              a.choice,
              a.correct ? 'نعم' : 'لا'
            ].forEach(v => {
              const td = document.createElement('td');

              td.textContent = String(v ?? '');

              tr.appendChild(td);
            });

            tbody.appendChild(tr);
          });

          table.appendChild(tbody);
          wrap.appendChild(table);
          card.appendChild(wrap);
          results.appendChild(card);
        });

      } catch (e) {
        status.textContent =
          'ما نجّمش نحمل القائمة. تأكّد اللي السيرفر يخدم.';

        console.error(e);
      }
    }

    document
      .getElementById('refresh')
      .addEventListener('click', loadList);

    loadList();
  </script>
